2. Add getNotificationPreferences function:
   ```typescript
   export async function getNotificationPreferences(userId: number): Promise<NotificationPreferences | null> {
     const user = await db.select().from(users).where(eq(users.id, userId)).get();
     if (!user?.notificationPreferences) return null;
     try {
       return JSON.parse(user.notificationPreferences) as NotificationPreferences;
     } catch {
       return null;
     }
   }
   ```

3. Add updateNotificationPreferences function:
   ```typescript
   export async function updateNotificationPreferences(
     userId: number,
     preferences: Partial<NotificationPreferences>
   ): Promise<boolean> {
     const existing = await getNotificationPreferences(userId);
     const updated = { ...existing, ...preferences };
     const result = await db
       .update(users)
       .set({ notificationPreferences: JSON.stringify(updated), updatedAt: new Date() })
       .where(eq(users.id, userId))
       .execute();
     return result.rowsAffected > 0;
   }
   ```

4. Add default preferences helper:
   ```typescript
   export function getDefaultPreferences(): NotificationPreferences {
     return {
       'order:created': true,
       'order:status-changed': true,
       'order:completed': true,
       'payment:received': true,
     };
   }
   ```

3. Add PUT /api/users/:id/notifications endpoint:
   ```typescript
   .put('/users/:id/notifications', async (ctx) => {
     const { id } = ctx.params;
     const body = await ctx.request.json().catch(() => ({}));
     const userId = parseInt(id);
     
     if (isNaN(userId)) return c.json({ error: 'Invalid user ID' }, 400);
     if (!body.preferences || typeof body.preferences !== 'object') {
       return c.json({ error: 'Missing preferences object' }, 400);
     }
     
     const success = await updateNotificationPreferences(userId, body.preferences);
     if (!success) return c.json({ error: 'User not found' }, 404);
     
     return c.json({ success: true, preferences: await getNotificationPreferences(userId) });
   })
   ```

   ```typescript
   async function shouldSendNotification(userId: number, eventType: string): Promise<boolean> {
     try {
       const prefs = await getNotificationPreferences(userId);
       const settings = prefs ?? getDefaultPreferences();
       return settings[eventType as keyof typeof settings] ?? true;
     } catch {
       // If check fails, allow notification (fail-open)
       return true;
     }
   }
   ```

   ```typescript
       const users = await getUsersByRoles(['chef', 'kasir']);
       for (const user of users) {
         if (await shouldSendNotification(user.id, 'order:created')) {
   ```

   ```typescript
     const users = await getUsersByRoles(roles);
     for (const user of users) {
   ```

   ```typescript
   // In navbar script or inline
   async function loadPreferences() {
     const res = await fetch('/api/users/me/notifications');
     const data = await res.json();
     const prefs = data.preferences;
     document.getElementById('pref-order:created').checked = prefs['order:created'];
     document.getElementById('pref-order:status-changed').checked = prefs['order:status-changed'];
     document.getElementById('pref-order:completed').checked = prefs['order:completed'];
     document.getElementById('pref-payment:received').checked = prefs['payment:received'];
   }
   
   async function savePreference(checkbox) {
     const prefKey = checkbox.id.replace('pref-', '');
     const prefValue = {};
     prefValue[prefKey] = checkbox.checked;
     await fetch('/api/users/me/notifications', {
       method: 'PUT',
       headers: { 'Content-Type': 'application/json' },
       body: JSON.stringify({ preferences: prefValue })
     });
   }
   ```
