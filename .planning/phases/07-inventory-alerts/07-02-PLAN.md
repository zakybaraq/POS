   ```typescript
   export async function checkStockLevels(
     ingredientId: number,
     io: Server
   ): Promise<boolean> {
     // Skip if already alerted
     if (alertedIngredients.has(ingredientId)) {
       return false;
     }
     
     // Get current stock and threshold
     const ingredient = await db.query.ingredients.findFirst({
       where: eq(ingredients.id, ingredientId)
     });
     
     if (!ingredient) return false;
     
     const currentStock = parseFloat(ingredient.currentStock);
     const threshold = parseFloat(ingredient.minStock);
     
     // Check if below threshold
     if (currentStock <= threshold) {
       await emitLowStockAlert(ingredient, io);
       alertedIngredients.add(ingredientId);
       return true;
     }
     
     return false;
   }
   ```

3. Implement emitLowStockAlert:
   ```typescript
   async function emitLowStockAlert(ingredient: Ingredient, io: Server) {
     const alert = {
       id: generateAlertId(),
       ingredientId: ingredient.id,
       ingredientName: ingredient.name,
       currentStock: ingredient.currentStock,
       threshold: ingredient.minStock,
       unit: ingredient.unit,
       timestamp: new Date().toISOString()
     };
     
     // Emit to admin and kitchen rooms
     io.to(ROOMS.ADMIN).to(ROOMS.KITCHEN).emit('inventory:low-stock', alert);
     
     // Log the alert
     logger.warn({
       ingredientId: ingredient.id,
       currentStock: ingredient.currentStock,
       threshold: ingredient.minStock
     }, 'Low stock alert triggered');
   }
   ```

4. Add acknowledgment function:
   ```typescript
   export function acknowledgeAlert(ingredientId: number): void {
     alertedIngredients.delete(ingredientId);
     logger.info({ ingredientId }, 'Alert acknowledged');
   }
   ```

   ```typescript
   export async function decrementStock(
     ingredientId: number,
     amount: number,
     orderId: number,
     io?: Server
   ): Promise<void> {
     await db.transaction(async (tx) => {
       // Get current stock
       const ingredient = await tx.query.ingredients.findFirst({
         where: eq(ingredients.id, ingredientId)
       });
       
       if (!ingredient) throw new Error('Ingredient not found');
       
       const currentStock = parseFloat(ingredient.currentStock);
       const newStock = currentStock - amount;
       
       // Update stock
       await tx.update(ingredients)
         .set({ currentStock: newStock.toString() })
         .where(eq(ingredients.id, ingredientId));
       
       // Record movement
       await tx.insert(stockMovements).values({
         ingredientId,
         type: 'out',
         quantity: amount.toString(),
         stockBefore: currentStock.toString(),
         stockAfter: newStock.toString(),
         referenceId: orderId,
         reason: 'Order fulfillment'
       });
     });
     
     // Check thresholds after decrement (outside transaction)
     if (io) {
       await checkStockLevels(ingredientId, io);
     }
   }
   ```

2. In src/websocket/events/inventory-events.ts, create event handlers:
   ```typescript
   import { Server, Socket } from 'socket.io';
   import { checkStockLevels, acknowledgeAlert } from '../../services/inventory-monitor';
   import { logger } from '../../logger';
   
   export function registerInventoryEvents(io: Server, socket: Socket) {
     // Client requests current low stock items
     socket.on('inventory:get-alerts', async () => {
       try {
         const lowStockItems = await getLowStockItems();
         socket.emit('inventory:alerts-list', lowStockItems);
       } catch (error) {
         logger.error({ error }, 'Failed to get low stock items');
         socket.emit('inventory:error', { message: 'Failed to fetch alerts' });
       }
     });
     
     // Client acknowledges an alert
     socket.on('inventory:acknowledge', (data: { ingredientId: number }) => {
       acknowledgeAlert(data.ingredientId);
       
       // Broadcast acknowledgment to all clients
       io.emit('inventory:alert-acknowledged', {
         ingredientId: data.ingredientId,
         acknowledgedAt: new Date().toISOString()
       });
       
       logger.info({ ingredientId: data.ingredientId }, 'Alert acknowledged via WebSocket');
     });
   }
   ```

3. Add getLowStockItems helper:
   ```typescript
   export async function getLowStockItems(): Promise<Ingredient[]> {
     const results = await db.query.ingredients.findMany({
       where: sql`CAST(current_stock AS DECIMAL) <= CAST(min_stock AS DECIMAL)`
     });
     return results;
   }
   ```

   ```javascript
   // Subscribe to alerts on socket connect
   socket.on('connect', () => {
     socket.emit('inventory:get-alerts');
   });
   
   // Handle incoming alerts
   socket.on('inventory:low-stock', (alert) => {
     addAlert(alert);
     playNotificationSound();
   });
   
   // Handle acknowledgment
   socket.on('inventory:alert-acknowledged', (data) => {
     removeAlert(data.ingredientId);
   });
   
   // Acknowledge button handler
   document.addEventListener('click', (e) => {
     if (e.target.classList.contains('ack-btn')) {
       const ingredientId = e.target.dataset.ingredientId;
       socket.emit('inventory:acknowledge', { ingredientId });
     }
   });
   
   function addAlert(alert) {
     const list = document.getElementById('alerts-list');
     const noAlertsMsg = list.querySelector('.no-alerts');
     if (noAlertsMsg) noAlertsMsg.remove();
     
     // Insert at top
     list.insertAdjacentHTML('afterbegin', renderAlert(alert));
     updateAlertCount();
   }
   
   function removeAlert(ingredientId) {
     const alertItem = document.querySelector(`[data-ingredient-id="${ingredientId}"]`);
     if (alertItem) {
       alertItem.remove();
       updateAlertCount();
     }
   }
   ```
